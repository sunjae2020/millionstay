export const spaceAvailability = pgTable('space_availability', {
  id: uuid('id').primaryKey().defaultRandom(),
  spaceId: uuid('space_id')
    .notNull()
    .references(() => spaces.id, { onDelete: 'cascade' }),
  date: date('date').notNull(),
  isAvailable: boolean('is_available').notNull().default(true),
  blockReason: varchar('block_reason', { length: 50 }),
  // values: 'Manual' | 'Booking' | 'iCal' | 'Maintenance'
  bookingId: uuid('booking_id')
    .references(() => bookings.id),
  notes: text('notes'),
  createdAt: timestamp('created_at', { withTimezone: true })
    .notNull()
    .defaultNow(),
}, (table) => ({
  // Unique: one record per space per date
  spaceIdDateUnique: unique().on(table.spaceId, table.date),
  // Index for fast date-range queries
  spaceIdDateIdx: index('idx_space_avail_space_date')
    .on(table.spaceId, table.date),
}));

Add these routes to the Express server:

1. GET /api/v1/space-availability/:spaceId
   Query: ?from=YYYY-MM-DD&to=YYYY-MM-DD (default: today + 30 days)
   Returns: array of { date, is_available, block_reason, booking_id }

2. POST /api/v1/space-availability/:spaceId/block
   Body: { dates: ["YYYY-MM-DD", ...], reason: "Manual" }
   Inserts/updates space_availability rows
   Returns: { blocked_count: N }

3. POST /api/v1/space-availability/:spaceId/unblock
   Body: { dates: ["YYYY-MM-DD", ...] }
   Sets is_available = true, clears block_reason and booking_id
   Returns: { unblocked_count: N }

4. GET /api/v1/system-logs
   Query: ?entity_type=&action=&from=&to=&limit=50&offset=0
   Returns: paginated system_log records
   (Read-only — no POST/PUT/DELETE)

5. GET /api/v1/email-templates
   Full CRUD

6. GET /api/v1/email-logs
   Query: ?entity_type=&entity_id=&limit=20
   Returns: paginated email_log records (Read-only)

7. GET /api/v1/recurring-schedules
   Full CRUD +
   POST /api/v1/recurring-schedules/generate
     Finds all active schedules where next_due_date <= today
     For each: creates invoice + invoice_product
     Updates next_due_date based on frequency:
       Weekly   → add 7 days
       Biweekly → add 14 days
       Monthly  → add 1 month
     Returns: { generated_count, invoice_ids: [] }

STEP 6 — Add system_log middleware

Create server/middleware/auditLog.ts (or .js):

export function auditLog(entityType: string) {
  return async (req, res, next) => {
    const originalJson = res.json.bind(res);
    res.json = async (data) => {
      if (data?.success && req.user) {
        const action = 
          req.method === 'POST'   ? 'CREATE' :
          req.method === 'PUT'    ? 'UPDATE' :
          req.method === 'DELETE' ? 'DELETE' :
          req.path.includes('/status') ? 'STATUS_CHANGE' : 'ACTION';
        
        await db.insert(systemLogs).values({
          entityType,
          entityId: req.params.id || data?.data?.id,
          action,
          actorType: 'User',
          actorId: req.user.id,
          actorEmail: req.user.email,
          newValue: req.body,
          ipAddress: req.ip,
        });
      }
      return originalJson(data);
    };
    next();
  };
}

Apply to key routes:
  router.post('/', auditLog('booking'), createBooking);
  router.put('/:id', auditLog('booking'), updateBooking);
  router.patch('/:id/confirm', auditLog('booking'), confirmBooking);
