SQL for the report endpoint:
SELECT
  b.booking_ref,
  c.first_name || ' ' || c.last_name AS guest_name,
  s.name AS space_name,
  b.check_in_date,
  b.check_out_date,
  CEIL((b.check_out_date::date -
        b.check_in_date::date) / 7.0) AS weeks,
  b.agreed_weekly_rate,
  ROUND(b.agreed_weekly_rate *
    CEIL((b.check_out_date::date -
          b.check_in_date::date) / 7.0), 2) AS total_rent,
  b.booking_status,
  b.booking_source
FROM bookings b
JOIN contacts c ON b.contact_id = c.id
JOIN spaces s   ON b.space_id   = s.id
WHERE b.is_active = true
  AND (b.check_in_date >= $1 OR $1 IS NULL)
  AND (b.check_in_date <= $2 OR $2 IS NULL)
  AND (b.booking_status = $3 OR $3 IS NULL)
ORDER BY b.created_at DESC
