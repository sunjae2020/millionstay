    account_id, schedule_type, frequency,
    amount, currency, gst_included,
    start_date, end_date (optional), next_due_date
  }

PUT    /api/v1/recurring-schedules/:id
PATCH  /api/v1/recurring-schedules/:id/deactivate
  Sets is_active = false

POST   /api/v1/recurring-schedules/generate-due
  Purpose: Cron trigger — generates invoices for overdue schedules
  Logic:
    1. SELECT all WHERE is_active=true AND next_due_date <= TODAY
    2. For each schedule:
       a. Create invoice:
          invoice_ref = 'MS-INV-' + YYYY + '-' + padded_sequence
          booking_id = schedule.booking_id
          account_id = schedule.account_id
          invoice_type = schedule.schedule_type
          subtotal = schedule.gst_included
                     ? ROUND(amount / 1.1, 2)
                     : amount
          gst_amount = schedule.gst_included
                       ? amount - subtotal
                       : 0
          total_amount = amount
          currency = schedule.currency
          issue_date = TODAY
          due_date = TODAY + 7 days
          invoice_status = 'Unpaid'

       b. Create invoice_product line:
          description = schedule.schedule_type + ' — ' +
                        frequency + ' payment'
          quantity = 1
          unit_price = subtotal
          gst_amount = schedule.gst_amount
          line_total = amount

       c. Update schedule:
          last_generated_at = NOW()
          next_due_date =
            frequency='Weekly'   → next_due_date + 7 days
            frequency='Biweekly' → next_due_date + 14 days
            frequency='Monthly'  → next_due_date + 1 month

    3. Return:
       { generated_count: N,
         invoice_refs: ["MS-INV-2026-00001", ...],
         errors: [] }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Create a reusable logAction() utility function:

// server/utils/auditLog.ts
export async function logAction({
  entityType,
  entityId,
  action,
  actorId,
  actorEmail,
  oldValue = null,
  newValue = null,
  ipAddress = null,
}: {
  entityType: string;
  entityId: number | string;
  action: 'CREATE'|'UPDATE'|'DELETE'|'STATUS_CHANGE'|
          'LOGIN'|'PAYMENT'|'VERIFY'|'BLOCK'|'UNBLOCK';
  actorId?: number;
  actorEmail?: string;
  oldValue?: object | null;
  newValue?: object | null;
  ipAddress?: string | null;
}) {
  try {
    await db.insert(systemLogs).values({
      entityType,
      entityId: String(entityId),
      action,
      actorType: actorId ? 'User' : 'System',
      actorId: actorId ? String(actorId) : null,
      actorEmail,
      oldValue: oldValue ? JSON.stringify(oldValue) : null,
      newValue: newValue ? JSON.stringify(newValue) : null,
      ipAddress,
    });
  } catch (err) {
    // Log failure should never crash the main request
    console.error('Audit log failed:', err);
  }
}

Apply logAction() to these critical operations:
  - bookings: CREATE, STATUS_CHANGE (confirm/cancel/checkin/checkout)
  - invoices: CREATE, STATUS_CHANGE (paid/void)
  - contracts: CREATE, STATUS_CHANGE
  - booking_documents: VERIFY, reject (action: 'VERIFY')
  - user login: action: 'LOGIN'
  - space_availability: BLOCK, UNBLOCK

Example usage in booking confirmation:
  await logAction({
    entityType: 'booking',
    entityId: booking.id,
    action: 'STATUS_CHANGE',
    actorId: req.user.id,
    actorEmail: req.user.email,
    oldValue: { status: 'PendingApproval' },
    newValue: { status: 'Confirmed' },
    ipAddress: req.ip,
  });
